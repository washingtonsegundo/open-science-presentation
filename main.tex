\begin{frame}{Four principles balance technical value and social responsibility}
  \begin{columns}[T,onlytextwidth]
    \column{.49\textwidth}
      \keypoint{FAIR}{Make data findable, accessible, interoperable, and reusable for people and machines.}
      \vspace{0.28cm}

      \keypoint{CARE}{Protect collective benefit, authority, responsibility, and ethics in data governance.}
    \column{.49\textwidth}
      \keypoint{TRUST}{Build transparent, responsible, user-focused, sustainable, and technically sound repositories.}
      \vspace{0.28cm}

      \keypoint{DEIA}{Embed diversity, equity, inclusion, and accessibility in research systems.}
  \end{columns}
  \vfill
  \begin{center}\small\textbf{Rule of thumb:} as open as possible, as closed as necessary.\end{center}
\end{frame}

      \tagline{What it adds to ARK}
      \vspace{0.24cm}
      \begin{itemize}\setlength\itemsep{0.30cm}
        \item \textbf{Decentralization:} participating institutions share operational responsibility.
        \item \textbf{Traceability:} identifier events remain auditable over time.
        \item \textbf{Resilience:} distributed governance reduces single points of failure.
        \item \textbf{Sovereignty:} regional institutions define policies and priorities.
      \end{itemize}
    \column{.45\textwidth}
      \tagline{How it works with Oasisbr}
      \vspace{0.24cm}
      \begin{enumerate}\setlength\itemsep{0.30cm}
        \item Repositories expose qualified metadata through open protocols.
        \item Oasisbr aggregates and validates the records.
        \item dARK assigns persistent identifiers to eligible objects.
        \item Users discover content centrally while access remains with the source institution.
      \end{enumerate}
  \end{columns}
