\begin{frame}{Publish, review, curate: the power of preprints}
  \begin{columns}[T,onlytextwidth]
    \column{.31\textwidth}
      \keypoint{1. Publish}{Share a preprint early, with a persistent identifier and clear version history.}
    \column{.31\textwidth}
      \keypoint{2. Review}{Invite critical evaluation and make reviews and author responses discoverable.}
    \column{.31\textwidth}
      \keypoint{3. Curate}{Select, contextualize, and recommend work using transparent editorial criteria.}
  \end{columns}
  \vspace{.35cm}
  \begin{itemize}\setlength\itemsep{.16cm}
    \item Earlier access can accelerate feedback, collaboration, and reuse.
    \item Link preprints, reviews, revised versions, and journal publications.
    \item Display review status clearly. Early availability does not establish validity.
  \end{itemize}
  \vfill
  {\tiny Source: \href{https://publish-review-curate.org/}{COAR and ASAPbio, Publish, Review, Curate initiative}.}
\end{frame}
